基础用法

<!-- 1. 添加显示元素 --> <span id="liveuser">加载中...</span> <!-- 2. 引入统计脚本 --> <script src="{{.ServerURL}}/liveuser.js"></script>

自定义配置

<!-- 指定站点ID --> <script src="{{.ServerURL}}/liveuser.js?siteId=my-blog"></script> <!-- 完整配置 --> <script src="{{.ServerURL}}/liveuser.js?siteId=my-site&displayElementId=counter&debug=true"></script>
